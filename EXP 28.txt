#include <stdio.h>
#include <string.h>

int main()
{
    char text[200], word[50];

    printf("Enter a sentence:\n");
    scanf(" %[^\n]", text);

    printf("Enter word to search: ");
    scanf("%s", word);

    if(strstr(text, word))
        printf("Word Found!\n");
    else
        printf("Word Not Found!\n");

    return 0;
}
