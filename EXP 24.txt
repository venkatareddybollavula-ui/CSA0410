#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int fd;
    char data[] = "Operating Systems Lab";

    fd = open("sample.txt", O_CREAT | O_WRONLY, 0644);

    if(fd < 0)
    {
        printf("File cannot be opened.\n");
        return 1;
    }

    write(fd, data, sizeof(data));

    printf("File created and data written successfully.\n");

    close(fd);

    return 0;
}
