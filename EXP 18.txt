#include <stdio.h>

int mutex=1, full=0, empty=3, x=0;

void producer()
{
    mutex--;
    empty--;
    full++;
    x++;
    printf("Producer produced item %d\n",x);
    mutex++;
}

void consumer()
{
    mutex--;
    full--;
    empty++;
    printf("Consumer consumed item %d\n",x);
    x--;
    mutex++;
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n1.Produce\n2.Consume\n3.Exit\n");
        scanf("%d",&ch);

        switch(ch)
        {
            case 1:
                if((mutex==1)&&(empty!=0))
                    producer();
                else
                    printf("Buffer Full\n");
                break;

            case 2:
                if((mutex==1)&&(full!=0))
                    consumer();
                else
                    printf("Buffer Empty\n");
                break;

            case 3:
                return 0;
        }
    }
}
